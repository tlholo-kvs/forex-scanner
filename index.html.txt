<script>
  const chartContainer = document.getElementById('chart-container');
  const chart = LightweightCharts.createChart(chartContainer, {
    layout: { background: { color: '#131722' }, textColor: '#d1d4dc' },
    grid: { vertLines: { color: '#1e222d' }, horzLines: { color: '#1e222d' } },
    timeScale: { timeVisible: true, secondsVisible: false },
    crosshair: { mode: LightweightCharts.CrosshairMode.Normal }
  });

  const candleSeries = chart.addCandlestickSeries({
    upColor: '#089981', downColor: '#f23645',
    borderVisible: false, wickUpColor: '#089981', wickDownColor: '#f23645'
  });

  let trendlineSeries = chart.addLineSeries({ color: '#2962ff', lineWidth: 2, title: 'Trendline' });
  let fibLines = [];
  let mode = null;
  let trendPoints = [];
  let fibPoints = [];
  let candleData = [];
  let allSetups = JSON.parse(localStorage.getItem('forexSetups') || '{}');

  function updateStatus(msg) {
    document.getElementById('status-bar').innerHTML = `Status: ${msg}`;
  }

  function setMode(newMode) {
    mode = newMode;
    document.getElementById('btnTrendline').classList.toggle('active-tool', mode === 'trendline');
    document.getElementById('btnFib').classList.toggle('active-tool', mode === 'fib');
    if (mode === 'trendline') {
      trendPoints = [];
      updateStatus('Click Point 1 of your Trendline on the chart.');
    } else if (mode === 'fib') {
      fibPoints = [];
      updateStatus('Click the Swing Low (or High) for your Fibonacci anchor.');
    }
  }

  async function loadChartData() {
    const apiKey = document.getElementById('apiKey').value.trim();
    const pair = document.getElementById('pairSelect').value;
    if (!apiKey) return alert('Please paste your Twelve Data API key first.');

    localStorage.setItem('twelveApiKey', apiKey);
    updateStatus(`Loading 1H candles for ${pair}...`);

    try {
      const url = `https://twelvedata.com{pair}&interval=1h&outputsize=120&apikey=${apiKey}`;
      const res = await fetch(url);
      const data = await res.json();

      if (data.status === 'error') {
        updateStatus(`API Error: ${data.message}`);
        return;
      }

      const precision = pair.includes('JPY') ? 3 : 5;
      const minMove = pair.includes('JPY') ? 0.001 : 0.00001;
      candleSeries.applyOptions({ priceFormat: { type: 'price', precision, minMove } });

      candleData = data.values.reverse().map(c => ({
        time: Math.floor(new Date(c.datetime + 'Z').getTime() / 1000),
        open: parseFloat(c.open),
        high: parseFloat(c.high),
        low: parseFloat(c.low),
        close: parseFloat(c.close)
      }));

      candleSeries.setData(candleData);
      chart.timeScale().fitContent();
      clearDrawings(false);
      updateStatus(`Loaded ${pair} (1H). Select a drawing tool above to mark your levels.`);
    } catch (err) {
      updateStatus('Failed to fetch chart data. Check your internet or API key.');
    }
  }

  chart.subscribeClick((param) => {
    if (!mode || !param.time || !param.point) return;
    const price = candleSeries.coordinateToPrice(param.point.y);
    const pair = document.getElementById('pairSelect').value;
    const decimals = pair.includes('JPY') ? 3 : 5;
    const cleanPrice = parseFloat(price.toFixed(decimals));

    if (mode === 'trendline') {
      trendPoints.push({ time: param.time, price: cleanPrice });
      if (trendPoints.length === 1) {
        updateStatus(`Trendline Point 1 set at ${cleanPrice}. Now click Point 2.`);
      } else if (trendPoints.length === 2) {
        trendPoints.sort((a, b) => a.time - b.time);
        drawExtendedTrendline();
        setMode(null);
        updateStatus('Trendline drawn! Now click "2. Draw Fibonacci".');
      }
    } else if (mode === 'fib') {
      fibPoints.push({ time: param.time, price: cleanPrice });
      if (fibPoints.length === 1) {
        updateStatus(`Fibonacci Anchor 1 set at ${cleanPrice}. Click Anchor 2.`);
      } else if (fibPoints.length === 2) {
        drawFibonacciLevels(decimals);
        setMode(null);
        updateStatus('Fibonacci Golden Zone plotted! Click "Save & Export JSON".');
      }
    }
  });

  function drawExtendedTrendline() {
    if (trendPoints.length < 2 || candleData.length === 0) return;
    const p1 = trendPoints[0];
    const p2 = trendPoints[1];
    const slope = (p2.price - p1.price) / (p2.time - p1.time);
    const lastCandleTime = candleData[candleData.length - 1].time;
    const futureTime = lastCandleTime + (3600 * 12); // Project 12 hours forward

    const lineData = [
      { time: p1.time, value: p1.price },
      { time: futureTime, value: p1.price + slope * (futureTime - p1.time) }
    ];
    trendlineSeries.setData(lineData);
  }

  function drawFibonacciLevels(decimals) {
    fibLines.forEach(line => candleSeries.removePriceLine(line));
    fibLines = [];

    const high = Math.max(fibPoints[0].price, fibPoints[1].price);
    const low = Math.min(fibPoints[0].price, fibPoints[1].price);
    const diff = high - low;
    const dir = document.getElementById('directionSelect').value;

    const fib50 = dir === 'BUY' ? high - (diff * 0.50) : low + (diff * 0.50);
    const fib618 = dir === 'BUY' ? high - (diff * 0.618) : low + (diff * 0.618);
    const fib786 = dir === 'BUY' ? high - (diff * 0.786) : low + (diff * 0.786);

    const levels = [
      { price: high, title: 'Swing High', color: '#787b86' },
      { price: fib50, title: '50.0% Fib', color: '#ff9800' },
      { price: fib618, title: '61.8% Golden', color: '#00e676' },
